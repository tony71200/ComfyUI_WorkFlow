import re


def _find_replace(text, find, replace, replace_in=None, whole_word=True):
    if replace_in is not None:
        if not replace:
            replace = replace_in
        elif "{in}" in replace:
            replace = replace.replace("{in}", replace_in)
    alts = [a.strip() for a in str(find or "").split("|") if a.strip()]
    if not alts:
        return text
    alts.sort(key=len, reverse=True)
    body = "|".join(re.escape(a) for a in alts)
    if whole_word:
        # "_" counts as a word char, so "tnga_brief" never matches "brief".
        body = rf"(?<!\w)(?:{body})(?!\w)"
    pattern = re.compile(body, re.IGNORECASE)
    return pattern.sub(lambda m: replace, text)


class FindAndReplace:
    @classmethod
    def INPUT_TYPES(cls):
        return {
            "required": {
                "text": ("STRING", {"forceInput": True}),
                "find": ("STRING", {"default": "", "multiline": False}),
                "replace": ("STRING", {"default": "", "multiline": False}),
                "whole_word": ("BOOLEAN", {"default": True, "tooltip": "Match whole words only (brief does not hit tnga_brief or briefly)."}),
            },
            "optional": {
                "replace_in": ("STRING", {"forceInput": True}),
            },
        }

    RETURN_TYPES = ("STRING",)
    RETURN_NAMES = ("text",)
    FUNCTION = "run"
    CATEGORY = "Tony4896/Text"

    def run(self, text, find, replace, whole_word=True, replace_in=None):
        return (_find_replace(text, find, replace, replace_in, whole_word),)

def split_txt(path):
    """Same split as TonyIO Text_Splitter with defaults (\\n\\n paragraphs, ### negative dropped)."""
    blocks = [b.strip() for b in Path(path).read_text(encoding="utf-8-sig").split("\n\n") if b.strip()]
    return [b.split("###", 1)[0].strip() for b in blocks]


def _named(wf, title):
    return next(n for n in wf["nodes"] if n.get("title") == title)["widgets_values_named"]


def _rx(wf, title, s):
    """Core RegexReplace."""
    v = _named(wf, title)
    flags = (re.I if v["case_insensitive"] else 0) | (re.M if v["multiline"] else 0) | (re.S if v["dotall"] else 0)
    return re.sub(v["regex_pattern"], v["replace"], s, count=v["count"], flags=flags)

```python
"""Wait until N new runs finish in ComfyUI /history, then dump the fields the v4 report needs.

usage: collect_history.py OUT.json N"""
import json
import sys
import time
import urllib.request
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent))
from pipeline_v4 import SWAP_ITEMS, TITLE  # noqa: E402

API = "http://127.0.0.1:8188"


def get(path):
    return json.load(urllib.request.urlopen(API + path, timeout=30))


def record(pid, h):
    api, extra = h["prompt"][2], h["prompt"][3]
    sel = next(k for k, v in api.items() if v.get("class_type") == "Select_Random")
    idx = api[sel]["inputs"]["index"]
    out = h.get("outputs", {})
    return {"prompt_id": pid, "ts_index": api["42"]["inputs"]["index"], "seed": api["39"]["inputs"]["value"],
            "nsfw": api["55"]["inputs"]["value"], "select_index": idx, "token": SWAP_ITEMS[min(idx, len(SWAP_ITEMS) - 1)],
            "final": "".join(out.get("44", {}).get("text", [""])),
            "images": [f'{i["subfolder"]}/{i["filename"]}' for i in out.get("46", {}).get("images", [])],
            "status": h.get("status", {}).get("status_str"), "api": api, "extra": extra}


if __name__ == "__main__":
    sys.stdout.reconfigure(encoding="utf-8")
    out_path, n = Path(sys.argv[1]), int(sys.argv[2])
    seen = set(get("/history"))
    done = {}
    while len(done) < n:
        time.sleep(5)
        for pid, h in get("/history").items():
            if pid not in seen and pid not in done and h.get("status", {}).get("completed") is not None:
                done[pid] = record(pid, h)
                r = done[pid]
                print(f"{len(done)}/{n} ts={r['ts_index']} sel={r['select_index']} {r['token']} {r['status']}", flush=True)
    runs = sorted(done.values(), key=lambda r: r["ts_index"])
    out_path.parent.mkdir(parents=True, exist_ok=True)
    out_path.write_text(json.dumps(runs, ensure_ascii=False, indent=2), encoding="utf-8")
    print("wrote", out_path)
```

Note: `"completed"` is a key of the history status object (true/false); a run that errored still ends with `status_str = "error"` and is recorded, so the collector does not hang on failures. Add `tools/krea2_v4/out/` to the repo `.gitignore`.

```python
"""Replay every pass-A run with only nsfw flipped to true (same seed, same TextSplitter and Select_Random index).

usage: replay_nsfw.py passA.json passB.json"""
import copy
import json
import sys
import time
import urllib.request
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent))
from collect_history import API, get, record  # noqa: E402


def post(path, body):
    req = urllib.request.Request(API + path, data=json.dumps(body).encode(), headers={"Content-Type": "application/json"})
    return json.load(urllib.request.urlopen(req, timeout=30))


if __name__ == "__main__":
    sys.stdout.reconfigure(encoding="utf-8")
    runs = json.load(open(sys.argv[1], encoding="utf-8"))
    out = []
    for r in runs:
        api = copy.deepcopy(r["api"])
        api["55"]["inputs"]["value"] = True
        api["46"]["inputs"]["filename_prefix"] += "_nsfw"
        pid = post("/prompt", {"prompt": api, "extra_data": r["extra"], "client_id": "krea2-v4-replay"})["prompt_id"]
        while True:
            time.sleep(5)
            h = get(f"/history/{pid}").get(pid)
            if h and h.get("status", {}).get("completed") is not None:
                break
        out.append(record(pid, h))
        print(f"{len(out)}/{len(runs)} ts={out[-1]['ts_index']} {out[-1]['status']}", flush=True)
    Path(sys.argv[2]).write_text(json.dumps(out, ensure_ascii=False, indent=2), encoding="utf-8")
    print("wrote", sys.argv[2])
```

"""Build the v4 test report tables from passA/passB.

usage: report_v4.py passA.json passB.json OUT.md"""
import json
import sys
from collections import Counter
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent))
from pipeline_v4 import SWAP_ITEMS, TEST_TXT, WF_PATH, check_final, simulate, split_txt  # noqa: E402

def fallback(wf, raw, r, prefix, suffix):
    """True when the final prompt equals the refine-off result, i.e. 4g fell back to the original prompt."""
    return r["final"] == simulate(wf, raw, prefix, suffix, r["token"])


if __name__ == "__main__":
    sys.stdout.reconfigure(encoding="utf-8")
    A, B = (json.load(open(p, encoding="utf-8")) for p in sys.argv[1:3])
    wf = json.loads(WF_PATH.read_text(encoding="utf-8"))
    raws = split_txt(TEST_TXT)
    pre = A[0]["api"]  # Prefix/Suffix values as queued
    prefix = next(v["inputs"]["value"] for v in pre.values() if v.get("_meta", {}).get("title") == "Prefix")
    suffix = next(v["inputs"]["value"] for v in pre.values() if v.get("_meta", {}).get("title") == "Suffix")
    L = ["## Kết quả tự động", "",
         f"- Pass A (SFW, Auto Batch): {len(A)} lượt, lỗi: {sum(r['status'] != 'success' for r in A)}",
         f"- Pass B (NSFW, phát lại): {len(B)} lượt, lỗi: {sum(r['status'] != 'success' for r in B)}",
         f"- Select_Random index pass A: {[r['select_index'] for r in A]} → phân bố {dict(Counter(r['token'] for r in A))}",
         f"- Fallback về prompt gốc: SFW {sum(fallback(wf, raws[r['ts_index']], r, prefix, suffix) for r in A)}, "
         f"NSFW {sum(fallback(wf, raws[r['ts_index']], r, prefix, suffix) for r in B)}", "",
         "| # | token | seed | SFW check | NSFW check | ảnh SFW | ảnh NSFW |", "|---|---|---|---|---|---|---|"]
    for a, b in zip(A, B):
        ca = ", ".join(check_final(a["final"], "Ivan_Ryo.", a["token"])) or "ok"
        cb = ", ".join(check_final(b["final"], "Ivan_Ryo.", b["token"])) or "ok"
        L.append(f"| {a['ts_index']} | {a['token']} | {a['seed']} | {ca} | {cb} | "
                 f"[png]({RESULT}{a['images'][0] if a['images'] else ''}) | [png]({RESULT}{b['images'][0] if b['images'] else ''}) |")
    L += ["", "## Prompt cuối từng lượt", ""]
    for a, b in zip(A, B):
        L += [f"### #{a['ts_index']} — {a['token']}", "", "**SFW:** " + a["final"], "", "**NSFW:** " + b["final"], ""]
    Path(sys.argv[3]).parent.mkdir(parents=True, exist_ok=True)
    Path(sys.argv[3]).write_text("\n".join(L), encoding="utf-8")
